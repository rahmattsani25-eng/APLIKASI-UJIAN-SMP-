A Server Google Spreadsheet Terpusat (Otomatis Pisah Sheet per Guru/Mapel)

Cukup buat 1 Google Spreadsheet Sekolah. Saat siswa mengumpulkan ujian, skrip di bawah ini otomatis membuat Tab/Sheet baru sesuai Kode Mapel Guru (misal: Sheet BIG-7, Sheet IPA-7, Sheet MTK-8).

Klik untuk Melihat Kode Google Apps Script Multi-Guru (Copy-Paste ke Google Sheets)
  1. Buka Google Drive sekolah > Buat Google Sheets baru bernama "SERVER NILAI UJIAN SEKOLAH".
  2. Klik menu Extensions (Ekstensi) > Apps Script.
  3. Ganti seluruh kode dengan kode di bawah ini, lalu klik Deploy > New deployment > Web app (Akses: Anyone / Siapa saja).
function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var data = JSON.parse(e.postData.contents);
  var roomCode = data.roomCode || "UMUM";

  // 1. Masukkan ke Sheet "MASTER_SEMUA_MAPEL"
  var master = ss.getSheetByName("MASTER_SEMUA_MAPEL");
  if (!master) {
    master = ss.insertSheet("MASTER_SEMUA_MAPEL");
    master.appendRow(["Waktu", "Kode Mapel", "Mata Pelajaran", "Guru", "Nama Siswa", "Kelas", "Absen", "Benar", "Salah", "Skor", "Durasi"]);
  }
  master.appendRow([data.timestamp, roomCode, data.examTitle, data.teacherName, data.name, data.className, data.absen, data.correctCount, data.wrongCount, data.score, data.duration]);

  // 2. Otomatis Buat / Isi Sheet Khusus Guru Mapel Tersebut (Misal: BIG-7, IPA-7)
  var roomSheet = ss.getSheetByName(roomCode);
  if (!roomSheet) {
    roomSheet = ss.insertSheet(roomCode);
    roomSheet.appendRow(["Waktu", "Nama Siswa", "Kelas", "No Absen", "Benar", "Salah", "Skor Akhir", "Durasi", "Mata Pelajaran", "Guru"]);
  }
  roomSheet.appendRow([data.timestamp, data.name, data.className, data.absen, data.correctCount, data.wrongCount, data.score, data.duration, data.examTitle, data.teacherName]);

  return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
}
